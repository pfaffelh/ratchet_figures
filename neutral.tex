\section*{Mean hitting time of $0$ for the diffusion
$\mathrm{d}X = -\lambda X\,\mathrm{d}t + \sqrt{X(1-X)}\,\mathrm{d}W$}

Let $X = (X_t)_{t \ge 0}$ on $[0,1]$ solve
\[
  \mathrm{d}X_t \;=\; -\lambda\,X_t\,\mathrm{d}t \;+\; \sqrt{X_t(1-X_t)}\,\mathrm{d}W_t,
  \qquad X_0 = x \in (0,1),
\]
with $\lambda > 0$. Let $\tau_0 := \inf\{t \ge 0 : X_t = 0\}$ and
$u(x) := \mathbb{E}_x[\tau_0]$.

\paragraph{ODE.} With $\sigma^2(x)=x(1-x)$ and drift $b(x)=-\lambda x$, the generator
is $\mathcal{L} = \tfrac12 x(1-x)\,\partial_x^2 - \lambda x\,\partial_x$, hence
\begin{equation}\label{eq:ode}
  \tfrac12\, x(1-x)\, u''(x) \;-\; \lambda x\, u'(x) \;=\; -1,
  \qquad u(0)=0.
\end{equation}
The right endpoint $x=1$ is non-exit (drift inward, diffusion vanishing); for
$2\lambda<1$ it is regular and we impose the reflecting condition $u'(1)=0$, for
$2\lambda\ge 1$ it is entrance and the only requirement is $u(1^-) <\infty$.

\paragraph{Scale and speed.} Since $2b/\sigma^2 = -2\lambda/(1-x)$,
\[
  S'(x) \;=\; \exp\!\left(-\!\int_0^x \tfrac{2b(y)}{\sigma^2(y)}\,\mathrm{d}y\right)
         \;=\; (1-x)^{-2\lambda},
  \qquad
  m(y) \;=\; \frac{2}{\sigma^2(y)\,S'(y)}
         \;=\; \frac{2\,(1-y)^{2\lambda-1}}{y}.
\]
The scale function is
\[
  S(x) \;=\; \int_0^x (1-w)^{-2\lambda}\,\mathrm{d}w \;=\;
  \begin{cases}
    \dfrac{\,1 - (1-x)^{1-2\lambda}\,}{1-2\lambda}, & 2\lambda \neq 1, \\[1.2ex]
    -\ln(1-x), & 2\lambda = 1.
  \end{cases}
\]

\paragraph{Solving \eqref{eq:ode}.} With $g := u'$,
\[
  g'(x) \;-\; \frac{2\lambda}{1-x}\,g(x) \;=\; -\,\frac{2}{x(1-x)}.
\]
Multiplying by the integrating factor $(1-x)^{2\lambda}$ and using that
$g(1^-)$ is finite,
\[
  (1-x)^{2\lambda}\,g(x)
  \;=\; 2\int_x^1 \frac{(1-y)^{2\lambda-1}}{y}\,\mathrm{d}y,
  \qquad
  u'(x) \;=\; 2\,(1-x)^{-2\lambda}\!\int_x^1 \frac{(1-y)^{2\lambda-1}}{y}\,\mathrm{d}y.
\]
Integrating from $0$ to $x$ and applying Fubini,
\begin{equation}\label{eq:result}
  \boxed{\;
    \mathbb{E}_x[\tau_0]
    \;=\;
    \int_0^x S(y)\,m(y)\,\mathrm{d}y
    \;+\;
    S(x)\int_x^1 m(y)\,\mathrm{d}y.
  \;}
\end{equation}

\paragraph{Regularity of the integrands.}
\begin{itemize}
\item As $y \to 0^+$, $S(y)\sim y$ and $m(y)\sim 2/y$, so $S(y)\,m(y)\to 2$
      (removable). The first integral in \eqref{eq:result} is thus over a
      bounded integrand.
\item For the second integral, the substitution $t=(1-y)^{2\lambda}$ gives
\begin{equation}\label{eq:I2sub}
  \int_x^1 m(y)\,\mathrm{d}y
  \;=\;
  \frac{1}{\lambda}\int_0^{(1-x)^{2\lambda}}\!\!\frac{\mathrm{d}t}{1 - t^{1/(2\lambda)}},
\end{equation}
whose integrand is bounded on $\bigl[0,(1-x)^{2\lambda}\bigr]$ for any $x>0$:
it equals $1$ at $t=0$ and $1/x$ at the upper limit.
\end{itemize}

\paragraph{Initial condition used in the visualization.} For the Muller's
ratchet figures we take
\[
  X_0 \;=\; e^{-\theta} \;=\; N^{-\delta},
  \qquad \theta = \delta \ln N,
\]
so that $u(e^{-\theta})$ is a function of $(\lambda, \theta)$ — i.e.\ of
$(\psi, \delta)$ via $\lambda = \psi\,\theta\,N^{-\beta}$,
$\theta = \delta \ln N$, $\beta = 1-\delta$. We evaluate
$u(e^{-\theta})$ numerically per parameter set using \eqref{eq:result}
with \eqref{eq:I2sub} and adaptive Simpson quadrature, and expose it as a
third theory baseline (``neutral diffusion: $x_0 = e^{-\theta}$'') in the
interactive viewer.
